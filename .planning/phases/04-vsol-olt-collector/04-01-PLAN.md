       ```python
       """
       Circuit breaker por OLT VSOL usando Redis TTL.

       VSOL-04: Suspende polling SSH 5 minutos tras 3 fallos consecutivos.
       Separado del circuit breaker Mikrotik (prefijo 'cb:').

       Claves Redis:
         vsol:fails:{device_id}  — contador de fallos (expira en CIRCUIT_OPEN_TTL)
         vsol:open:{device_id}   — indica circuit abierto (TTL = tiempo de suspension)
       """
       import redis.asyncio as aioredis

       _PREFIX = "vsol"
       CIRCUIT_OPEN_TTL: int = 5 * 60   # 5 minutos (VSOL-04)
       CIRCUIT_FAIL_THRESHOLD: int = 3  # 3 fallos SSH consecutivos


       async def is_vsol_circuit_open(redis_client: aioredis.Redis, device_id: int) -> bool:
           """True si el circuit breaker VSOL esta abierto para esta OLT."""
           return await redis_client.exists(f"{_PREFIX}:open:{device_id}") > 0


       async def record_vsol_failure(redis_client: aioredis.Redis, device_id: int) -> bool:
           """
           Incrementa contador de fallos SSH VSOL.
           Si alcanza CIRCUIT_FAIL_THRESHOLD, abre el circuit (setex con TTL).
           Retorna True si el circuit acaba de abrirse.
           """
           fail_key = f"{_PREFIX}:fails:{device_id}"
           count = await redis_client.incr(fail_key)
           await redis_client.expire(fail_key, CIRCUIT_OPEN_TTL)

           if count >= CIRCUIT_FAIL_THRESHOLD:
               await redis_client.setex(f"{_PREFIX}:open:{device_id}", CIRCUIT_OPEN_TTL, "1")
               await redis_client.delete(fail_key)
               return True
           return False


       async def record_vsol_success(redis_client: aioredis.Redis, device_id: int) -> None:
           """Resetea el circuit breaker VSOL al conectar exitosamente."""
           await redis_client.delete(f"{_PREFIX}:fails:{device_id}")
           await redis_client.delete(f"{_PREFIX}:open:{device_id}")
       ```

       ```python
       """
       Parsers de CLI VSOL V2.3.1R para recoleccion de ONUs GPON.

       VSOL-01: parse_onu_state extrae estado y serial por ONU en un puerto.
       VSOL-01: parse_optical_info extrae Rx/Tx dBm de una ONU especifica.

       CRITICO — Confianza MEDIUM:
       Los formatos de salida documentados aqui estan basados en VSOL V2.x documentation
       knowledge. Deben validarse contra el hardware real BeepyRed_OLT_GPON_SanLuis
       (firmware V2.3.1R) en la primera ejecucion. Los parsers loggean la salida raw
       para facilitar el diagnostico si el regex no coincide.

       Estrategia defensiva: regex con re.search() (no posicion de columna) para
       tolerar variaciones de espaciado entre versiones de firmware.
       """
       import logging
       import re
       from typing import Any

       logger = logging.getLogger(__name__)

       # Regex para linea de ONU en "show pon onu state gpon 0/X"
       # Formato esperado: gpon-onu_0/PORT:ONU_NUM  SERIAL  SN  STATE  ...
       # Ejemplo:          gpon-onu_0/1:1           4857454c12345678  SN  ONLINE  OK  1234
       _ONU_LINE_RE = re.compile(
           r"gpon-onu_0/(\d+):(\d+)"   # grupo 1=port, grupo 2=onu_index
           r"\s+(\w{16})"              # grupo 3=serial (16 chars hex)
           r"\s+\S+"                   # auth-type (ignorar: SN, LOID, etc.)
           r"\s+(ONLINE|OFFLINE|RANGING)"  # grupo 4=state
       )

       # Regex para "show pon optical-info gpon 0/X onu N"
       _RX_RE = re.compile(r"[Rr]x\s+power\s*\(dBm\)\s*:\s*(-?[\d.]+)")
       _TX_RE = re.compile(r"[Tt]x\s+power\s*\(dBm\)\s*:\s*(-?[\d.]+)")


       def parse_onu_state(stdout: str, port: int) -> list[dict[str, Any]]:
           """
           Parsea salida de 'show pon onu state gpon 0/PORT'.

           Retorna lista de dicts con campos:
             - onu_index (int): numero de ONU dentro del puerto
             - serial_number (str): 16 chars hex
             - state (str): "ONLINE", "OFFLINE" o "RANGING"
             - port (int): numero de puerto GPON (1-8)

           Devuelve lista vacia si no se encuentran ONUs o si el stdout esta vacio.
           Loggea la salida raw en DEBUG para facilitar diagnostico hands-on.
           """
           if not stdout or not stdout.strip():
               return []

           logger.debug("vsol_parser: raw state output port %d:\n%s", port, stdout)

           onus = []
           for line in stdout.splitlines():
               m = _ONU_LINE_RE.search(line)
               if m:
                   onus.append({
                       "port": int(m.group(1)),
                       "onu_index": int(m.group(2)),
                       "serial_number": m.group(3),
                       "state": m.group(4),
                   })

           if not onus:
               logger.warning(
                   "vsol_parser: parse_onu_state port %d — regex no coincidio con ninguna linea. "
                   "Verificar formato CLI del hardware real. stdout (primeros 300 chars):\n%s",
                   port, stdout[:300],
               )

           return onus


       def parse_optical_info(stdout: str) -> dict[str, float] | None:
           """
           Parsea salida de 'show pon optical-info gpon 0/X onu N'.

           Retorna dict con claves "signal_rx_dbm" y/o "signal_tx_dbm" (float).
           Retorna None si no se encuentran los valores esperados.

           Solo llamar para ONUs con state=ONLINE — para OFFLINE/RANGING el OLT
           puede retornar output vacio o error.
           """
           if not stdout or not stdout.strip():
               return None

           logger.debug("vsol_parser: raw optical output:\n%s", stdout)

           result: dict[str, float] = {}

           m_rx = _RX_RE.search(stdout)
           if m_rx:
               try:
                   result["signal_rx_dbm"] = float(m_rx.group(1))
               except ValueError:
                   pass

           m_tx = _TX_RE.search(stdout)
           if m_tx:
               try:
                   result["signal_tx_dbm"] = float(m_tx.group(1))
               except ValueError:
                   pass

           return result if result else None
       ```

       ```python
       async def _collect_state_async(device_id: int) -> dict:
           """Logica async: SSH al OLT, poll de estado por puerto, upsert de ONUs."""
           redis_client = aioredis.from_url(settings.REDIS_URL, decode_responses=True)
           try:
               # VSOL-04: verificar circuit breaker ANTES de conectar
               if await is_vsol_circuit_open(redis_client, device_id):
                   logger.info("OLT %s: vsol circuit open, skipping SSH poll", device_id)
                   return {"skipped": True, "reason": "circuit_open", "device_id": device_id}

               olt, credentials = await _get_olt_and_credentials(device_id)
               if olt is None:
                   return {"skipped": True, "reason": "device_not_found", "device_id": device_id}
               if olt.device_type != DeviceType.OLT_VSOL_GPON:
                   return {"skipped": True, "reason": "not_vsol_gpon", "device_id": device_id}
               if credentials is None:
                   logger.warning("OLT %s: no SSH credential found", device_id)
                   return {"skipped": True, "reason": "no_credential", "device_id": device_id}

               username, password = credentials
               all_onus = await _fetch_onu_states(olt.ip_address, username, password, device_id, redis_client)

               if all_onus is None:
                   return {"collected": False, "device_id": device_id}

               # Upsert all ONUs in a single session
               async with AsyncSessionLocal() as db:
                   for onu_data in all_onus:
                       await upsert_onu(
                           db=db,
                           olt_device_id=device_id,
                           pon_port=f"gpon 0/{onu_data['port']}",
                           serial_number=onu_data["serial_number"],
                           onu_status=onu_data["state"],
                           onu_index=onu_data["onu_index"],
                       )
                   await db.commit()

               return {"collected": True, "device_id": device_id, "onus_processed": len(all_onus)}

           finally:
               await redis_client.aclose()


       async def _fetch_onu_states(
           host: str,
           username: str,
           password: str,
           device_id: int,
           redis_client: aioredis.Redis,
       ) -> list[dict] | None:
           """
           Abre UNA sesion SSH al OLT y ejecuta 8 comandos de estado secuencialmente.
           VSOL-03: async with asyncssh.connect() garantiza cierre en finally.
           """
           try:
               # VSOL-03: connect_timeout y login_timeout aplican timeout duro
               async with asyncssh.connect(
                   host,
                   username=username,
                   password=password,
                   known_hosts=None,                  # red privada ISP — no verificar host key
                   connect_timeout=settings.VSOL_SSH_TIMEOUT,
                   login_timeout=settings.VSOL_SSH_TIMEOUT,
               ) as conn:
                   all_onus: list[dict] = []
                   for port in range(1, 9):  # gpon 0/1 a gpon 0/8
                       result = await conn.run(
                           f"show pon onu state gpon 0/{port}",
                           timeout=settings.VSOL_SSH_TIMEOUT,
                       )
                       port_onus = parse_onu_state(result.stdout or "", port)
                       all_onus.extend(port_onus)

               await record_vsol_success(redis_client, device_id)
               return all_onus

           except Exception as exc:
               logger.error("OLT %s: SSH error: %s", device_id, exc)
               just_opened = await record_vsol_failure(redis_client, device_id)
               if just_opened:
                   logger.warning("OLT %s: vsol circuit breaker opened (3 consecutive SSH failures)", device_id)
               return None

       async def _collect_optical_async(device_id: int) -> dict:
           """Logica async: SSH al OLT, optical-info por ONU ONLINE, actualizar senal."""
           redis_client = aioredis.from_url(settings.REDIS_URL, decode_responses=True)
           try:
               if await is_vsol_circuit_open(redis_client, device_id):
                   return {"skipped": True, "reason": "circuit_open", "device_id": device_id}

               olt, credentials = await _get_olt_and_credentials(device_id)
               if olt is None or olt.device_type != DeviceType.OLT_VSOL_GPON:
                   return {"skipped": True, "reason": "device_not_found_or_wrong_type"}
               if credentials is None:
                   return {"skipped": True, "reason": "no_credential"}

               # Obtener ONUs ONLINE de esta OLT
               async with AsyncSessionLocal() as db:
                   result = await db.execute(
                       select(ONU)
                       .where(ONU.olt_id == device_id)
                       .where(ONU.onu_status == "ONLINE")
                   )
                   online_onus = list(result.scalars().all())

               if not online_onus:
                   return {"collected": True, "device_id": device_id, "optical_collected": 0}

               username, password = credentials
               optical_data = await _fetch_optical_data(
                   olt.ip_address, username, password, online_onus, device_id, redis_client
               )

               if optical_data is None:
                   return {"collected": False, "device_id": device_id}

               # Actualizar senal en DB
               async with AsyncSessionLocal() as db:
                   for onu_id, signal in optical_data.items():
                       result = await db.execute(select(ONU).where(ONU.id == onu_id))
                       onu = result.scalar_one_or_none()
                       if onu and signal:
                           onu.signal_rx_dbm = signal.get("signal_rx_dbm")
                           onu.signal_tx_dbm = signal.get("signal_tx_dbm")
                           onu.last_updated_at = datetime.now(timezone.utc)
                   await db.commit()

               return {"collected": True, "device_id": device_id, "optical_collected": len(optical_data)}

           finally:
               await redis_client.aclose()


       async def _fetch_optical_data(
           host: str,
           username: str,
           password: str,
           online_onus: list[ONU],
           device_id: int,
           redis_client: aioredis.Redis,
       ) -> dict[int, dict] | None:
           """
           Una sola sesion SSH — optical-info por ONU ONLINE.
           Solo ejecuta el comando para ONUs con onu_status=ONLINE.
           """
           try:
               async with asyncssh.connect(
                   host,
                   username=username,
                   password=password,
                   known_hosts=None,
                   connect_timeout=settings.VSOL_SSH_TIMEOUT,
                   login_timeout=settings.VSOL_SSH_TIMEOUT,
               ) as conn:
                   optical: dict[int, dict] = {}
                   for onu in online_onus:
                       if not onu.pon_port:
                           continue
                       # pon_port formato: "gpon 0/1" -> port=1; onu_index from onu record
                       port_num = onu.pon_port.split("/")[-1]
                       # Obtener onu_index de pon_port en onus table
                       # El onu_index se almacena en pon_port? No — necesitamos el indice.
                       # Por ahora, parsear el indice del pon_port o usar una convencion.
                       # NOTA: el onu_index dentro del puerto se necesita para el comando.
                       # Se resuelve almacenando el onu_index en el campo pon_port extendido.
                       # Ver upsert_onu: pon_port = "gpon 0/PORT:ONU_INDEX"
                       if ":" in onu.pon_port:
                           onu_idx = onu.pon_port.split(":")[-1]
                           port_num = onu.pon_port.split("/")[-1].split(":")[0]
                       else:
                           continue  # formato invalido, saltar

                       result = await conn.run(
                           f"show pon optical-info gpon 0/{port_num} onu {onu_idx}",
                           timeout=settings.VSOL_SSH_TIMEOUT,
                       )
                       parsed = parse_optical_info(result.stdout or "")
                       if parsed:
                           optical[onu.id] = parsed

               await record_vsol_success(redis_client, device_id)
               return optical

           except Exception as exc:
               logger.error("OLT %s: SSH optical error: %s", device_id, exc)
               await record_vsol_failure(redis_client, device_id)
               return None

       # Helpers
       # ---------------------------------------------------------------------------

       async def _get_olt_and_credentials(
           device_id: int,
       ) -> tuple[Device | None, tuple[str, str] | None]:
           """Retorna (Device, (username, password)) o (None, None) si no existe."""
           async with AsyncSessionLocal() as db:
               result = await db.execute(select(Device).where(Device.id == device_id))
               device = result.scalar_one_or_none()
               if device is None:
                   return None, None

               cred_result = await db.execute(
                   select(DeviceCredential)
                   .where(DeviceCredential.device_id == device_id)
                   .where(DeviceCredential.credential_type == "ssh")
                   .limit(1)
               )
               cred = cred_result.scalar_one_or_none()
               if cred is None or cred.encrypted_password is None:
                   return device, None

               password = decrypt_credential(cred.encrypted_password)
               return device, (cred.username or "", password)


       async def upsert_onu(
           db: AsyncSession,
           olt_device_id: int,
           pon_port: str,        # formato "gpon 0/PORT:ONU_INDEX" ej: "gpon 0/1:1"
           serial_number: str,
           onu_status: str,      # "ONLINE", "OFFLINE", "RANGING"
           onu_index: int,
       ) -> None:
           """
           VSOL-05: Crea o actualiza Device + ONU record para una ONU.

           Si serial_number no existe en tabla onus: crea Device (type=ONU) + ONU.
           Si ya existe: actualiza onu_status y last_updated_at.
           NO hace commit — el caller acumula todos los cambios y hace un solo commit.
           """
           # Buscar ONU existente por serial_number
           result = await db.execute(
               select(ONU).where(ONU.serial_number == serial_number)
           )
           onu_record = result.scalar_one_or_none()

           # Estado del device GPON -> status del inventario
           device_status = (
               DeviceStatus.UP if onu_status == "ONLINE" else DeviceStatus.DOWN
           )
           now = datetime.now(timezone.utc)
           full_pon_port = f"gpon 0/{pon_port.split('/')[-1].split(':')[0]}:{onu_index}" if ":" not in pon_port else pon_port

           if onu_record is None:
               # Crear Device para la ONU (ip_address="0.0.0.0" — no tiene IP routable)
               onu_name = f"ONU {serial_number[:8]}"
               new_device = Device(
                   name=onu_name,
                   ip_address="0.0.0.0",
                   device_type=DeviceType.ONU,
                   parent_device_id=olt_device_id,
                   pon_port=full_pon_port,
                   status=device_status,
                   is_active=True,
                   last_seen_at=now if onu_status == "ONLINE" else None,
               )
               db.add(new_device)
               await db.flush()  # obtener ID

               onu_record = ONU(
                   device_id=new_device.id,
                   olt_id=olt_device_id,
                   serial_number=serial_number,
                   pon_port=full_pon_port,
                   onu_status=onu_status,
                   last_updated_at=now,
               )
               db.add(onu_record)

           else:
               # Actualizar estado existente
               onu_record.onu_status = onu_status
               onu_record.last_updated_at = now

               # Actualizar Device.status
               result2 = await db.execute(
                   select(Device).where(Device.id == onu_record.device_id)
               )
               dev = result2.scalar_one_or_none()
               if dev:
                   dev.status = device_status
                   if onu_status == "ONLINE":
                       dev.last_seen_at = now
       ```

## Trust Boundaries

| Boundary | Descripción |
|----------|-------------|
| Celery worker -> SSH OLT VSOL | Credenciales SSH descifradas en runtime; datos CLI del OLT entran al worker |
| CLI output -> DB | Texto no estructurado del OLT se parsea antes de insertar en onus/devices |
| asyncssh known_hosts=None | Red privada — se acepta cualquier host key sin verificación |

## STRIDE Threat Register

| Threat ID | Categoría | Componente | Disposición | Mitigación |
|-----------|-----------|------------|-------------|------------|
| T-4-01 | Information Disclosure | `_get_olt_and_credentials` — password SSH en memoria | mitigate | No loggear username ni password. `decrypt_credential()` solo en runtime. Solo loggear device_id. |
| T-4-02 | Tampering | `parse_onu_state` — serial_number de CLI inyectado en DB | mitigate | serial_number es solo string descriptivo; máx 100 chars en columna (String(100)). No ejecutable. Parseo defensivo con regex. |
| T-4-03 | Denial of Service | Máx 3 sesiones SSH al OLT — circuit breaker protege | mitigate | Circuit breaker abre tras 3 fallos. Si técnico tiene sesión manual abierta, circuit breaker absorbe los intentos fallidos sin spam. |
| T-4-04 | Spoofing | known_hosts=None — sin verificación de host key | accept | Red privada ISP — riesgo MITM negligible. Para v2: capturar y persistir host key en primera conexión. |
| T-4-05 | Information Disclosure | Redis keys `vsol:open:{id}` | accept | Redis interno Railway. Solo contienen flags y contadores — sin datos sensibles. |
| T-4-06 | Denial of Service | 597 ONUs × upsert × commit único | accept | Un solo commit por ciclo de polling del OLT — no 597 commits individuales. SQLAlchemy usa executemany/batch donde posible. |
