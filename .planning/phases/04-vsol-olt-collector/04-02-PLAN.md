    ```python
    async def _check_signal_alert(
        db: AsyncSession,
        redis_client: aioredis.Redis,
        onu_device_id: int,
        onu_name: str,
        pon_port: str,
        signal_rx_dbm: float,
        olt_site: str | None,
    ) -> None:
        """
        Verifica si la senal optica ONU esta por debajo del umbral configurado.
        Envia alerta Telegram si cae por debajo del umbral y el debounce no esta activo.

        Debounce: clave Redis 'vsol:signal_alert:{onu_device_id}' con TTL 3600s (1 hora).
        Evita spam de alertas si la senal oscila alrededor del umbral.
        """
        from app.services.thresholds import get_threshold
        from app.services.telegram import send_telegram_alert

        threshold = await get_threshold(db, "signal_low", device_id=onu_device_id)
        if threshold is None or signal_rx_dbm >= threshold:
            return

        # Verificar debounce — no reenviar si ya se envio en la ultima hora
        debounce_key = f"vsol:signal_alert:{onu_device_id}"
        if await redis_client.exists(debounce_key):
            return

        # Enviar alerta y establecer debounce
        text = (
            f"📡 <b>SEÑAL ONU BAJA</b>\n"
            f"<b>ONU:</b> {onu_name}\n"
            f"<b>Puerto:</b> {pon_port}\n"
            f"<b>Rx:</b> <code>{signal_rx_dbm:.2f} dBm</code> "
            f"(umbral: {threshold:.1f} dBm)\n"
            f"<b>Sitio:</b> {olt_site or 'Sin sitio'}"
        )
        await send_telegram_alert(text)
        await redis_client.setex(debounce_key, 3600, "1")
        logger.info(
            "ONU device %s: signal alert sent (%.2f dBm < %.1f dBm threshold)",
            onu_device_id, signal_rx_dbm, threshold,
        )
    ```

       ```python
       """
       Endpoint de inventario de ONUs GPON.

       VSOL-05: Lista ONUs con su OLT padre, puerto PON, estado y senal optica.
       Protegido con JWT — mismo patron que /incidents (Phase 3).

       Filtros opcionales:
         - olt_id: solo ONUs de esa OLT
         - pon_port: solo ONUs en ese puerto (ej: "gpon 0/1:1" o "gpon 0/1")
         - onu_status: solo ONUs con ese estado ("ONLINE", "OFFLINE", "RANGING")
         - limit/offset: paginacion
       """
       from typing import Any

       from fastapi import APIRouter, Depends, Query
       from sqlalchemy import select
       from sqlalchemy.ext.asyncio import AsyncSession

       from app.api.deps import get_current_user, get_db
       from app.models.device import Device
       from app.models.onu import ONU
       from app.schemas.onu import ONUResponse

       router = APIRouter(tags=["onus"])


       @router.get("/onus", response_model=list[ONUResponse])
       async def list_onus(
           olt_id: int | None = Query(default=None),
           pon_port: str | None = Query(default=None),
           onu_status: str | None = Query(default=None),
           limit: int = Query(default=100, le=500),
           offset: int = Query(default=0, ge=0),
           db: AsyncSession = Depends(get_db),
           _: dict = Depends(get_current_user),
       ) -> list[dict[str, Any]]:
           """
           Lista ONUs GPON con senal optica y estado actual.
           Requiere JWT valido.
           """
           query = (
               select(
                   ONU.id.label("onu_id"),
                   ONU.device_id,
                   ONU.olt_id,
                   ONU.serial_number,
                   ONU.pon_port,
                   ONU.onu_status,
                   ONU.signal_rx_dbm,
                   ONU.signal_tx_dbm,
                   ONU.last_updated_at,
                   Device.name.label("device_name"),
                   Device.site,
               )
               .join(Device, ONU.device_id == Device.id)
               .order_by(ONU.olt_id, ONU.pon_port)
               .limit(limit)
               .offset(offset)
           )

           if olt_id is not None:
               query = query.where(ONU.olt_id == olt_id)
           if pon_port is not None:
               query = query.where(ONU.pon_port == pon_port)
           if onu_status is not None:
               query = query.where(ONU.onu_status == onu_status)

           result = await db.execute(query)
           # result.mappings().all() es el patron correcto para JOINs con columnas individuales
           # (scalars() no funciona cuando se seleccionan columnas individuales, no modelos completos)
           return [dict(row) for row in result.mappings().all()]
       ```

       ```tsx
                       <TableHead>Puerto</TableHead>
                       <TableHead>Serial</TableHead>
       ```
